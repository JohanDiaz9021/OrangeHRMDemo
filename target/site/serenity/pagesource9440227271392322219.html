  <script>

    exigirSesion();

    pintarEncabezado('reservar');



    const mensaje = document.querySelector('[data-testid="mensaje-reserva"]');

    const selectMascota = document.querySelector('[name="mascota"]');

    const campoEntrada = document.getElementById('fecha-entrada');

    const campoSalida = document.querySelector('[data-testid="fecha-salida"]');

    const checkBano = document.getElementById('srv-bano');

    const checkPaseo = document.querySelector('[name="paseo"]');

    const panelResumen = document.querySelector('.panel-resumen');

    const botonConfirmar = panelResumen.querySelector('.btn-primario');

    let calculoActual = null;



    document.querySelector('[data-testid="fecha-hoy"]').textContent = textoFecha(hoyUtc());



    function mostrarMensaje(texto, tipo) {

      mensaje.innerHTML = texto;

      mensaje.className = 'mensaje ' + (tipo || 'exito');

      mensaje.hidden = false;

    }



    function cargarMascotas() {

      const mascotas = obtenerMascotas();

      document.querySelector('.sin-mascotas').hidden = mascotas.length > 0;

      document.querySelector('.disposicion-dos').hidden = mascotas.length === 0;

      selectMascota.innerHTML = '<option value="">Selecciona una mascota</option>' +

        mascotas.map(m => `<option value="${m.id}">${escaparHtml(m.nombre)} (${m.especie}, ${m.tamano})</option>`).join('');



      const preseleccion = new URLSearchParams(window.location.search).get('mascota');

      if (preseleccion && mascotaPorId(preseleccion)) {

        selectMascota.value = preseleccion;

      }

      actualizarPaseo();

    }



    function actualizarPaseo() {

      const mascota = mascotaPorId(selectMascota.value);

      const esGato = mascota && mascota.especie === 'Gato';

      checkPaseo.disabled = esGato;

      if (esGato) checkPaseo.checked = false;

      document.querySelector('.nota-paseo').hidden = !esGato;

    }



    function invalidarResumen() {

      calculoActual = null;

      panelResumen.hidden = true;

    }



    function validar() {

      const errores = {};

      const mascota = mascotaPorId(selectMascota.value);

      const entrada = leerFecha(campoEntrada.value);

      const salida = leerFecha(campoSalida.value);

      const hoy = hoyUtc();



      if (!mascota) errores.mascota = 'Selecciona una mascota';



      if (!campoEntrada.value.trim()) {

        errores.entrada = 'Este campo es obligatorio';

      } else if (entrada === null) {

        errores.entrada = 'Usa el formato DD/MM/AAAA con una fecha válida';

      } else if (entrada < hoy) {

        errores.entrada = 'La fecha de entrada no puede ser anterior a hoy';

      }



      if (!campoSalida.value.trim()) {

        errores.salida = 'Este campo es obligatorio';

      } else if (salida === null) {

        errores.salida = 'Usa el formato DD/MM/AAAA con una fecha válida';

      } else if (entrada !== null && salida <= entrada) {

        errores.salida = 'La fecha de salida debe ser posterior a la de entrada';

      } else if (entrada !== null && diasEntre(entrada, salida) > MAXIMO_NOCHES) {

        errores.salida = 'La estadía máxima es de 30 noches';

      }



      if (Object.keys(errores).length === 0) {

        const cruce = obtenerReservas().some(r =>

          r.mascotaId === mascota.id && r.estado === 'Confirmada' &&

          entrada < r.salida && r.entrada < salida

        );

        if (cruce) errores.general = 'Esta mascota ya tiene una reserva activa en esas fechas';

      }



      document.querySelectorAll('.error-campo').forEach(span => {

        span.textContent = errores[span.dataset.campo] || '';

      });



      return { errores, mascota, entrada, salida };

    }



    function calcular(mascota, entrada, salida) {

      const noches = diasEntre(entrada, salida);

      const tarifa = TARIFAS[mascota.tamano];

      const alojamiento = noches * tarifa;

      const descuento = noches >= NOCHES_PARA_DESCUENTO ? Math.round(alojamiento * DESCUENTO_ESTADIA_LARGA) : 0;

      const bano = checkBano.checked ? PRECIO_BANO : 0;

      const paseo = checkPaseo.checked ? PRECIO_PASEO : 0;

      const total = alojamiento - descuento + bano + paseo;

      return { noches, tarifa, alojamiento, descuento, bano, paseo, total };

    }



    document.getElementById('btn-calcular').addEventListener('click', function () {

      mensaje.hidden = true;

      const r = validar();

      if (Object.keys(r.errores).length > 0) {

        invalidarResumen();

        mostrarMensaje(r.errores.general || 'Revisa los campos marcados', 'error');

        return;

      }



      const c = calcular(r.mascota, r.entrada, r.salida);

      calculoActual = { ...c, mascota: r.mascota, entrada: r.entrada, salida: r.salida };



      panelResumen.querySelector('.res-mascota').textContent = r.mascota.nombre;

      panelResumen.querySelector('[data-testid="res-noches"]').textContent = c.noches;

      panelResumen.querySelector('.res-tarifa').textContent = formatoPesos(c.tarifa);

      document.getElementById('res-alojamiento').textContent = formatoPesos(c.alojamiento);

      panelResumen.querySelector('.fila-descuento').hidden = c.descuento === 0;

      panelResumen.querySelector('[data-testid="res-descuento"]').textContent = formatoPesos(-c.descuento);

      panelResumen.querySelector('.fila-bano').hidden = c.bano === 0;

      panelResumen.querySelector('.res-bano').textContent = formatoPesos(c.bano);

      panelResumen.querySelector('.fila-paseo').hidden = c.paseo === 0;

      document.getElementById('res-paseo').textContent = formatoPesos(c.paseo);

      panelResumen.querySelector('[data-testid="res-total"]').textContent = formatoPesos(c.total);

      panelResumen.hidden = false;

    });



    botonConfirmar.addEventListener('click', async function () {

      if (!calculoActual) return;

      botonConfirmar.disabled = true;

      mostrarCargando('Reservando la habitación...');

      await esperaAleatoria(800, 1500);

      ocultarCargando();

      botonConfirmar.disabled = false;



      const reservas = obtenerReservas();

      const codigo = generarCodigo('PS-', reservas.map(r => r.codigo));

      reservas.push({

        codigo: codigo,

        mascotaId: calculoActual.mascota.id,

        mascota: calculoActual.mascota.nombre,

        entrada: calculoActual.entrada,

        salida: calculoActual.salida,

        noches: calculoActual.noches,

        bano: calculoActual.bano > 0,

        paseo: calculoActual.paseo > 0,

        total: calculoActual.total,

        estado: 'Confirmada',

        reembolso: 0

      });

      guardar(CLAVES.reservas, reservas);



      document.getElementById('form-reserva').reset();

      actualizarPaseo();

      invalidarResumen();

      mostrarMensaje('Reserva confirmada con el código <strong data-testid="codigo-reserva">' + codigo +

        '</strong>. <a href="reservas.html">Ver mis reservas</a>', 'exito');

    });



    [selectMascota, campoEntrada, campoSalida, checkBano, checkPaseo].forEach(control => {

      control.addEventListener('change', invalidarResumen);

    });

    selectMascota.addEventListener('change', actualizarPaseo);



    cargarMascotas();

  </script>
